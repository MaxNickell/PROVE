# PROVE Pipeline Implementation Plan
# Programmatic Reasoning Over Visual Evidence
# Claude Code Step-by-Step Implementation Guide

## OVERVIEW
Build a minimal end-to-end pipeline that answers comparative questions for two images by:
1. Collecting image evidence (objects, attributes, intra relations, inter comparisons)
2. Converting evidence into probabilistic facts
3. Running ProbLog queries
4. Returning open-ended answer + short explanation

class PROVEPipeline:
    """Main PROVE pipeline for comparative visual reasoning."""
    
    def __init__(self):
        self.evidence_collector = EvidenceCollector()
        self.reasoning_engine = ReasoningEngine()
        
    def run(self, image_a_path: str, image_b_path: str, question: str) -> KnowledgeBase:
        """Run complete PROVE pipeline."""
        
        # Initialize knowledge base
        kb = KnowledgeBase(question)
        
        # Collect evidence from both images
        kb = self.evidence_collector.collect_evidence(kb, image_a_path, image_b_path)
        
        # Generate probabilistic reasoning and answer
        kb = self.reasoning_engine.generate_answer(kb)
        
        return kb
    
    def run_and_save(self, image_a_path: str, image_b_path: str, 
                     question: str, output_path: str) -> Dict:
        """Run pipeline and save results to JSON."""
        kb = self.run(image_a_path, image_b_path, question)
        result_json = kb.to_json()
        
        with open(output_path, 'w') as f:
